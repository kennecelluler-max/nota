/* ── Animasi angka statistik ── */
function countUp(el, target, money){
  const from = Number(el.dataset.v || 0); el.dataset.v = target;
  const start = performance.now(), dur = 550;
  (function f(t){
    const p = Math.min((t - start) / dur, 1), e = 1 - Math.pow(1 - p, 3);
    const val = Math.round(from + (target - from) * e);
    el.textContent = money ? rupiah(val) : val;
    if (p < 1) requestAnimationFrame(f);
  })(start);
}

/* ═══════════ NOMOR NOTA ═══════════ */
function nextNomor(){
  const d = new Date();
  const ymd = String(d.getFullYear()).slice(2) + pad(d.getMonth() + 1) + pad(d.getDate());
  let seq = DB.get("ns_seq", { ymd, n: 0 });
  if (seq.ymd !== ymd) seq = { ymd, n: 0 };
  seq.n++; DB.set("ns_seq", seq);
  return `NS-${ymd}-${String(seq.n).padStart(3, "0")}`;
}

function resetFormFields(){
  ["fNama","fWa","fMerek","fWarna","fImei","fPola","fKeluhan","fDiskon","fDp","itNama","itHarga"].forEach(i => $("#"+i).value = "");

  formItems = []; editingId = null;

function openForm(nota){
  resetFormFields();
  if (nota){
    editingId = nota.id;
    $("#formTitle").textContent = "Edit Nota";
    $("#formBadge").textContent = nota.nomor;
    $("#fNama").value = nota.cust.nama; $("#fWa").value = nota.cust.wa || "";
    $("#fMerek").value = nota.device.merek || ""; $("#fWarna").value = nota.device.warna || "";
    $("#fImei").value = nota.device.imei || ""; $("#fPola").value = nota.device.pola || "";

      <div class="ir-name">${esc(it.nama)}<small>${rupiah(it.harga)} × ${it.qty}</small></div>

});

function updateTotals(){
  const sub   = formItems.reduce((a, i) => a + i.qty * i.harga, 0);
  const diskon = parseAngka($("#fDiskon").value), dp = parseAngka($("#fDp").value);
  const total = Math.max(sub - diskon, 0), sisa = Math.max(total - dp, 0);
  $("#tSub").textContent = rupiah(sub);
  $("#tDis").textContent = "− " + rupiah(diskon);
  $("#tTot").textContent = rupiah(total);
  $("#tDp").textContent  = rupiah(dp);
  $("#tSisa").textContent = rupiah(sisa);
}
["fDiskon","fDp","itHarga"].forEach(id => {
  $("#"+id).addEventListener("input", updateTotals);
  $("#"+id).addEventListener("blur", e => { const n = parseAngka(e.target.value); e.target.value = n ? n.toLocaleString("id-ID") : ""; });
});

function waText(n){
  const s = settings, L = [];
  L.push(`*${s.nama.toUpperCase()}* — Nota Servis`, `No: ${n.nomor}`, `🗓 ${fmtTgl(n.tgl)} ${fmtJam(n.tgl)}`, "");
  L.push(`👤 ${n.cust.nama}`, `📱 ${n.device.merek || "-"}${n.device.warna ? " • " + n.device.warna : ""}`);
  if (n.keluhan) L.push(`📝 Keluhan: ${n.keluhan}`);
  L.push("", "Rincian:");
  n.items.forEach(i => L.push(`• ${i.nama} ${i.qty}x ${rupiah(i.harga)}`));
  L.push("", `Total: ${rupiah(n.total)}`, `DP: ${rupiah(n.dp)}`, `*Sisa: ${rupiah(n.sisa)}*`, `Status: ${n.status.toUpperCase()}`);
  if (n.estimasi) L.push(`Estimasi selesai: ${fmtTgl(n.estimasi)}`);
  L.push("", s.footer);
  return L.join("\n");
}

    if (await confirmBox("Hapus nota?", `Nota ${n.nomor} (${n.cust.nama}) akan dihapus permanen.`, "Hapus")){
      notas = notas.filter(x => x.id !== n.id); persist(); toast("Nota dihapus"); renderHistory();
    }
  }
});

let sKat = "Semua";
function renderSvcCats(){
  const cats = ["Semua", ...KATEGORI_LAYANAN];
  $("#svcCats").innerHTML = cats.map(k => `<button class="fchip ${k===sKat?"active":""}" data-k="${k}">${k}</button>`).join("");
  $$("#svcCats .fchip").forEach(c => c.onclick = () => { sKat = c.dataset.k; renderServices(); });
}
$("#sSearch").addEventListener("input", renderServices);
function renderServices(){
  renderSvcCats();
  const q = $("#sSearch").value.toLowerCase();
  const list = services.filter(s =>
    (sKat === "Semua" || s.kategori === sKat) &&
    (!q || s.nama.toLowerCase().includes(q) || s.kategori.toLowerCase().includes(q))
  );
  const box = $("#svcList");
  box.innerHTML = list.length ? list.map(s => `
    <div class="svc-row reveal">
      <div class="sn"><b>${esc(s.nama)}</b><span>${esc(s.kategori)}</span></div>
      <span class="price mono">${rupiah(s.harga)}</span>
      <button class="icon-btn" data-act="edit" data-id="${s.id}">✏</button>
      <button class="icon-btn danger" data-act="del" data-id="${s.id}">🗑</button>
    </div>`).join("") : `<div class="empty"><span class="big">🔧</span><p>Belum ada layanan di kategori ini.</p></div>`;
  observe(box);
}
$("#svcList").addEventListener("click", async e => {
  const b = e.target.closest("button[data-act]"); if (!b) return;
  const s = services.find(x => x.id === b.dataset.id); if (!s) return;
  if (b.dataset.act === "edit") openSvcModal(s);
  if (b.dataset.act === "del" && await confirmBox("Hapus layanan?", `"${s.nama}" akan dihapus dari daftar.`, "Hapus")){
    services = services.filter(x => x.id !== s.id); persist(); buildDatalist(); renderServices(); toast("Layanan dihapus");
  }
});
function openSvcModal(s){
  $("#svcModalTitle").textContent = s ? "Edit Layanan" : "Tambah Layanan";
  $("#svId").value = s ? s.id : "";
  $("#svNama").value = s ? s.nama : "";
  $("#svKat").value = s ? s.kategori : "";
  $("#svHarga").value = s ? s.harga.toLocaleString("id-ID") : "";
  $("#svcModal").classList.add("show");
}

$("#btnSvcSave").onclick = () => {
  const nama = $("#svNama").value.trim(), kat = $("#svKat").value.trim() || "Lainnya", harga = parseAngka($("#svHarga").value);
  if (!nama || !harga){ toast("Nama & harga wajib diisi ⚠"); return; }
  const id = $("#svId").value;
  if (id){ const s = services.find(x => x.id === id); Object.assign(s, { nama, kategori: kat, harga }); }

  persist(); buildDatalist(); renderServices();
  $("#svcModal").classList.remove("show"); toast("Layanan disimpan ✓");
};

$("#importFile").addEventListener("change", e => {
  const f = e.target.files[0]; if (!f) return;
  const r = new FileReader();
  r.onload = async () => {
    try{
      const d = JSON.parse(r.result);
      if (!Array.isArray(d.notas)) throw 0;
      if (await confirmBox("Import backup?", `${d.notas.length} nota & ${(d.services||[]).length} layanan akan MENGGANTI data saat ini.`, "Import")){
        notas = d.notas; services = d.services?.length ? d.services : services;
        if (d.settings) settings = { ...settings, ...d.settings };
        persist(); buildDatalist(); applyIdentity(); go("home"); toast("Backup berhasil diimport ✓");
      }
    }catch(err){ toast("File tidak valid ⚠"); }
    e.target.value = "";
  };
  r.readAsText(f);
});
